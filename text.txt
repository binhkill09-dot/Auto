local function getPlayerMoney()
    local function checkFolder(folder)
        if not folder then return nil end
        local cashNames = {"Cash", "Money", "Coins", "Tokens"}
        for _, name in ipairs(cashNames) do
            local val = folder:FindFirstChild(name)
            if val and (val:IsA("IntValue") or val:IsA("NumberValue")) then
                return val.Value
            end
        end
        return nil
    end
    
    local money = checkFolder(localPlayer:FindFirstChild("leaderstats")) or checkFolder(localPlayer)
    return money or -1 
end

local function getTowerLevel(tower)
    if not tower then return 1 end
    local lvlObj = tower:FindFirstChild("Level") or tower:FindFirstChild("Upgrade")
    if lvlObj and (lvlObj:IsA("IntValue") or lvlObj:IsA("NumberValue")) then
        return lvlObj.Value
    end
    local attr = tower:GetAttribute("Level") or tower:GetAttribute("Upgrade")
    if attr then return tonumber(attr) end
    return 1 
end

local function getCurrentWave()
    local playerGui = localPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return 1 end

    for _, gui in ipairs(playerGui:GetDescendants()) do
        if gui:IsA("TextLabel") or gui:IsA("TextButton") then
            local text = gui.Text
            if text and text ~= "" then
                if string.find(text:lower(), "wave") then
                    local num = tonumber(string.match(text, "%d+"))
                    if num and num > 0 and num < 150 then
                        return num
                    end
                end
            end
        end
    end

    for _, gui in ipairs(playerGui:GetDescendants()) do
        if (gui:IsA("TextLabel") or gui:IsA("TextButton")) and gui.AbsolutePosition.Y < 250 then
            local text = gui.Text
            local num = tonumber(string.match(text, "^%s*(%d+)%s*$"))
            if num and num > 0 and num < 150 then
                return num
            end
        end
    end

    return 1
end

local function placeAndGetTowerSafe(name, cframe)
    local oldTowers = workspace.Towers:GetChildren()
    local placed = false
    
    while not placed do
        local success, canPlace = pcall(function()
            return RemoteFunctions.CanPlaceTower:InvokeServer(name)
        end)
        if success and canPlace then
            local pSuccess = pcall(function()
                RemoteFunctions.PlaceTower:InvokeServer(name, cframe)
            end)
            if pSuccess then
                placed = true
            end
        else
            task.wait(1)
        end
    end
    
    while true do
        for _, tower in ipairs(workspace.Towers:GetChildren()) do
            if tower.Name == name and not table.find(oldTowers, tower) then
                return tower
            end
        end
        task.wait(0.3)
    end
end

local function upgradeTowerBulletproof(tower, costTable)
    if not tower then return end
    local totalLevels = #costTable
    
    for level = 1, totalLevels do
        local cost = costTable[level]
        local upgraded = false
        local attempts = 0
        
        while not upgraded do
            if not tower or not tower.Parent then return end
            attempts = attempts + 1
            
            if attempts > 60 then
                pcall(function() RemoteFunctions.UpgradeTower:InvokeServer(tower) end)
                attempts = 0
            end
            
            local currentCash = getPlayerMoney()
            
            if currentCash >= 0 then
                if currentCash >= cost then
                    pcall(function() RemoteFunctions.UpgradeTower:InvokeServer(tower) end)
                    task.wait(1.5)
                    upgraded = true
                else
                    task.wait(1)
                end
            else
                local oldLevel = getTowerLevel(tower)
                pcall(function() RemoteFunctions.UpgradeTower:InvokeServer(tower) end)
                task.wait(2)
                local newLevel = getTowerLevel(tower)
                
                if newLevel > oldLevel then
                    upgraded = true
                else
                    task.wait(1)
                end
            end
        end
    end
end

-- Hàm nâng cấp siêu tốc trong vòng 2 giây
local function superFastUpgradeTower(tower, costTable)
    if not tower then return end
    for level = 1, #costTable do
        local upgraded = false
        local startTime = tick()
        while not upgraded and (tick() - startTime < 2) do
            if not tower or not tower.Parent then return end
            pcall(function()
                RemoteFunctions.UpgradeTower:InvokeServer(tower)
            end)
            task.wait(0.05)
            if getTowerLevel(tower) > level then
                upgraded = true
            end
        end
    end
end
