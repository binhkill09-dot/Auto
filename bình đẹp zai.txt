local function getPlayerMoney()
    local function checkFolder(folder)
        if not folder then return nil end
        local cashNames = {"Cash", "Money", "Coins", "Tokens"}
        for _, name in ipairs(cashNames) do
            local val = folder:FindFirstChild(name)
            if val and (val:IsA("IntValue") or val:IsA("NumberValue")) then
                return val.Value
            end
        end
        return nil
    end
    
    local money = checkFolder(localPlayer:FindFirstChild("leaderstats")) or checkFolder(localPlayer)
    return money or -1 
end

local function getTowerLevel(tower)
    local lvlObj = tower:FindFirstChild("Level") or tower:FindFirstChild("Upgrade")
    if lvlObj and (lvlObj:IsA("IntValue") or lvlObj:IsA("NumberValue")) then
        return lvlObj.Value
    end
    local attr = tower:GetAttribute("Level") or tower:GetAttribute("Upgrade")
    if attr then return tonumber(attr) end
    return 1 
end

-- Đã chỉnh sửa: Tất cả Titan đều trả về đúng một vị trí cố định (TitanBase)
local function getSafeTitanCFrame(index)
    return POSITIONS.TitanBase
end

local function placeAndGetTowerSafe(name, cframe)
    print("[Bình AI]: Đang chờ đặt " .. name .. "...")
    
    local oldTowers = workspace.Towers:GetChildren()
    
    local placed = false
    while not placed do
        local success, canPlace = pcall(function()
            return RemoteFunctions.CanPlaceTower:InvokeServer(name)
        end)
        if success and canPlace then
            local pSuccess = pcall(function()
                RemoteFunctions.PlaceTower:InvokeServer(name, cframe)
            end)
            if pSuccess then
                placed = true
                print("-> [Thành công] Đã gửi lệnh đặt: " .. name)
            end
        else
            task.wait(0.5)
        end
    end
    
    while true do
        for _, tower in ipairs(workspace.Towers:GetChildren()) do
            if tower.Name == name and not table.find(oldTowers, tower) then
                print("-> [Thành công] Đã khóa mục tiêu tháp mới!")
                return tower
            end
        end
        task.wait(0.3)
    end
end

local function upgradeTitanBulletproof(tower)
    if not tower then return end
    print("[Bình AI]: Bắt đầu tiến trình Up MAX 3 cấp!")
    
    for level = 1, 3 do
        local cost = UPGRADE_COSTS[level]
        local upgraded = false
        print("[Bình AI]: Đang chờ đủ " .. cost .. "$ để lên cấp " .. level .. "...")
        
        while not upgraded do
            if not tower or not tower.Parent then return end
            
            local currentCash = getPlayerMoney()
            
            if currentCash >= 0 then
                if currentCash >= cost then
                    pcall(function() RemoteFunctions.UpgradeTower:InvokeServer(tower) end)
                    upgraded = true
                    print("-> Đã up cấp " .. level)
                    task.wait(1)
                else
                    task.wait(1) 
                end
            else
                local oldLevel = getTowerLevel(tower)
                pcall(function() RemoteFunctions.UpgradeTower:InvokeServer(tower) end)
                task.wait(2) 
                local newLevel = getTowerLevel(tower)
                
                if newLevel > oldLevel then
                    upgraded = true
                    print("-> Đã up cấp " .. level)
                else
                    task.wait(1) 
                end
            end
        end
    end
    print("[Bình AI]: Hoàn thành MAX Level! Chuyển con tiếp theo.")
end

local function sellAllTowers(name)
    print("[Bình AI]: Đang bán toàn bộ " .. name)
    for _, tower in ipairs(workspace.Towers:GetChildren()) do
        if tower.Name == name then
            pcall(function() RemoteFunctions.SellTower:InvokeServer(tower) end)
            task.wait(0.2)
        end
    end
end
