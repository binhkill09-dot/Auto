local capturedData = {}

-- 3. Hàm xử lý dữ liệu an toàn (chống lặp vô hạn gây đứng game)
local function serialize(v, depth)
    depth = depth or 0
    if depth > 6 then return '"[Giới hạn bảng]"' end -- Chống kẹt nếu table quá sâu

    local t = typeof(v)
    if t == "string" then return string.format("%q", v)
    elseif t == "number" or t == "boolean" then return tostring(v)
    elseif t == "Instance" then
        if not v.Parent then return 'nil --[[' .. v.Name .. ']]' end
        local path, curr = {}, v
        pcall(function()
            while curr and curr ~= game do
                table.insert(path, 1, curr.Name)
                curr = curr.Parent
            end
        end)
        if #path == 0 then return 'game["' .. v.Name .. '"]' end
        return 'game:GetService("' .. path[1] .. '").' .. table.concat(path, ".", 2)
    elseif t == "CFrame" then return "CFrame.new(" .. tostring(v) .. ")"
    elseif t == "Vector3" then return "Vector3.new(" .. tostring(v) .. ")"
    elseif t == "table" then
        local str = "{"
        for k, val in pairs(v) do
            str = str .. "[" .. serialize(k, depth + 1) .. "] = " .. serialize(val, depth + 1) .. ", "
        end
        return str .. "}"
    else
        return "nil --[[" .. t .. "]]"
    end
end

-- Hàm lưu log ngầm
local function SaveLog(remote, method, args)
    pcall(function()
        local path = serialize(remote)
        local argsCode = ""
        for i, arg in ipairs(args) do
            argsCode = argsCode .. serialize(arg) .. (i < #args and ", " or "")
        end
        local line = string.format("-- %s\nlocal args = {%s}\n%s:%s(unpack(args))\n", remote.Name, argsCode, path, method)
        table.insert(capturedData, line)
        ExportBtn.Text = "Export Logs (" .. #capturedData .. ")"
    end)
end

-- 4. Hook Namecall (Sử dụng task.spawn để không bao giờ làm kẹt thao tác trong game)
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    
    if method == "FireServer" or method == "InvokeServer" then
        local args = {...}
        -- Đẩy việc lưu log sang một luồng khác, trả lại luồng chính cho game ngay lập tức
        task.spawn(function()
            if typeof(self) == "Instance" then
                SaveLog(self, method, args)
            end
        end)
    end
    
    return oldNamecall(self, ...)
end)

-- 5. Nút Export
ExportBtn.MouseButton1Click:Connect(function()
    if #capturedData == 0 then
        ExportBtn.Text = "Chưa có data!"
        task.wait(1)
        ExportBtn.Text = "Export Logs (0)"
        return
    end
    
    local finalOutput = "--[[ \n    SimpleSpy Auto-Captured Logs\n    Total: " .. #capturedData .. "\n]]\n\n" .. table.concat(capturedData, "\n----------------------------------------\n\n")
    
    if setclipboard then
        setclipboard(finalOutput)
        ExportBtn.Text = "Đã Copy!"
        ExportBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
    end
    
    task.wait(2)
    ExportBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    ExportBtn.Text = "Export Logs (" .. #capturedData .. ")"
end)
